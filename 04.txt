                  $select.val(columnId);

                function revertSelect($select) {
                  $select.val($select.data('current-column-id'));
                }
              
                // Exposto globalmente para os modais existentes chamarem no success
                window.triggerFunnelMove = function (callback) {
                  if (!pendingMove) {
                    if (callback) callback();
                    return;
                  }
              
                  var move = pendingMove;
                  pendingMove = null;
                  window.triggerFunnelMove = null;
              
                  $.ajax({
                    url: funnelBaseUrl(move.funnelId, move.leadId) + '/move',
                    method: 'PUT',
                    headers: { 'X-CSRF-Token': csrfToken() },
                    data: { target_column_id: move.targetColumnId, force: true },
                    success: function () {
                      if (callback) callback();
                    },
                    error: function (xhr) {
                      var msg = (xhr.responseJSON && xhr.responseJSON.error) || 'Erro ao mover coluna do funil';
                      toastr.error(msg);
                      if (callback) callback();
                    }
                  });
                };
              
                function openExistingModal(modalType, move) {
                  if (modalType === 'visit_scheduling' || modalType === 'unified_visit' || modalType === 'visit_confirmation') {
                    // Carrega o modal de atividade existente — abre visit_confirmation se houver visita agendada
                    $('.loading').show();
                    $.get('/webapp/leads/' + move.leadId + '/activity?only_visit=true')
                      .always(function () { $('.loading').hide(); });
              
                  } else if (modalType === 'closing') {
                    $('#modal_converted').modal('show');
              
                  } else if (modalType === 'orientative') {
                    // Move direto sem modal — já estamos no lead, interação suficiente
                    $.ajax({
                      url: funnelBaseUrl(move.funnelId, move.leadId) + '/move',
                      method: 'PUT',
                      headers: { 'X-CSRF-Token': csrfToken() },
                      data: { target_column_id: move.targetColumnId, force: true },
                      success: function () {
                        updateSelect(move.$select, move.targetColumnId);
                        toastr.success('Lead movido com sucesso');
                        pendingMove = null;
                        window.triggerFunnelMove = null;
                      },
                      error: function (xhr) {
                        revertSelect(move.$select);
                        var msg = (xhr.responseJSON && xhr.responseJSON.error) || 'Erro ao mover lead';
                        toastr.error(msg);
                      }
                    });

            const cancelButton = target.closest(`${this.selectors.cancelButtonFilter}, .cancel-button`);
            const pillButton = target.closest(this.selectors.filterPillButton);
            const neighborhoodBadge = target.closest('.neighborhood-badge');

            if (insertButton) { e.preventDefault(); this._insertPropertiesIntoMessage(insertButton); return; }

              e.preventDefault();
              $('#modal_indicator').modal('hide');
              return;
            }
            if (pillButton) {
              e.preventDefault();
              this._switchResultsView('original');
              return;
            }
            if (target.closest('.custom-select-trigger')) { this._handleDropdownToggle(target.closest('.custom-select-trigger')); return; }
            if (target.closest('.custom-select-option')) { this._handleDropdownSelection(target.closest('.custom-select-option'), e); return; }
            if (!target.closest('.custom-select-wrapper')) { this._closeAllDropdowns(); }
            if (target.classList.contains('remove-item-filter')) { this._removeSelectedItem(target); }
            if (target.closest('.rooms-content .selected, .parking-content .selected')) { this._handleRoomParkingSelection(target.closest('.rooms-content .selected, .parking-content .selected')); }

              options.forEach(opt => {
                const labelText = opt.querySelector('label').textContent.toLowerCase();

          const maxValueInput = document.querySelector(this.selectors.maxValueInput);

          if (maxValueInput) maxValueInput.addEventListener('input', () => this._validatePriceRange());
      
          document.querySelector(this.selectors.advancedSearchButton)?.addEventListener('click', () => this._toggleAdvancedSearch(true));
          document.querySelector(this.selectors.backButton)?.addEventListener('click', () => this._toggleAdvancedSearch(false));
          document.querySelector(this.selectors.searchButton)?.addEventListener('click', e => { e.preventDefault(); this.searchProperties(); });

              scrollContainer.addEventListener('scroll', this._handleScroll.bind(this));

        _fillPriceRangeInputs() {

          const maxPrice = Math.round(basePrice * 1.3);

            priceInputs[0].value = minPrice.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ".");

        _validatePriceRange() {
          const minValueInput = document.querySelector(this.selectors.minValueInput);
          const maxValueInput = document.querySelector(this.selectors.maxValueInput);
          const errorContainer = document.querySelector(this.selectors.priceErrorContainer);
          const searchButton = document.querySelector(this.selectors.searchButton);
      
          if (!minValueInput || !maxValueInput || !errorContainer || !searchButton) return;
      
          const getNumericValue = (value) => {
            if (!value) return 0;
            return parseInt(value.replace(/\./g, ""), 10) || 0;
          };
      
          const minValue = getNumericValue(minValueInput.value);
          const maxValue = getNumericValue(maxValueInput.value);
      
          if (maxValue > 0 && minValue > maxValue) {
            errorContainer.innerHTML = `<span style="color: red; font-size: 12px; display: block; margin-top: 4px;">O valor mínimo não pode ser superior ao valor máximo.</span>`;
            errorContainer.style.display = 'block';
            
            searchButton.disabled = true;
            searchButton.style.cursor = 'not-allowed';
            searchButton.style.backgroundColor = '#A7A6A6';
          } else {
            errorContainer.textContent = '';
            errorContainer.style.display = 'none';
      
            searchButton.disabled = false;
            searchButton.style.cursor = 'pointer';
            searchButton.style.backgroundColor = '';
          }
        },
      
        _getFilterState() {
          const getNumericValue = (val) => val ? val.replace(/\./g, "") : null;
          const getSelectedMulti = (type) => Array.from(document.querySelectorAll(`.custom-select-wrapper[data-type="${type}"] .selected-item-filter`)).map(item => item.dataset.value);
          const getSelectedCheckboxes = (selector) => Array.from(document.querySelectorAll(`${selector}:checked`)).map(cb => cb.value);
      
          return {
            text: document.querySelector('.search')?.value.trim() || null,
            transaction_types: getSelectedCheckboxes(this.selectors.negotiationCheckboxes),
            neighborhoods: getSelectedMulti('neighborhood'),
            property_subtypes: getSelectedMulti('property-type'),
            bedrooms: this.state.selectedRooms,
            parking_spots: this.state.selectedParkingSpaces,
            price_range: { min: getNumericValue(document.querySelector('.value-range .min-value')?.value), max: getNumericValue(document.querySelector('.value-range .max-value')?.value) },
            area_range: { min: getNumericValue(document.querySelector('.total-area .min-area')?.value), max: getNumericValue(document.querySelector('.total-area .max-area')?.value) }
          };
        },
      
        _buildFilterUrl(baseUrl, filters, page) {
          const params = new URLSearchParams();
          params.append('page', page);
          params.append('per_page', 10);
      
          Object.entries(filters).forEach(([key, value]) => {
            if (value === null || value === '') return;
      
            if (Array.isArray(value)) {
              if (value.length > 0) {
                value.forEach(v => params.append(`filters[${key}][]`, v));
              } else if (key === 'transaction_types') {
                params.append(`filters[${key}][]`, '');
              }
            } else if (typeof value === 'object') {
              if (value.min) params.append(`filters[${key}][min]`, value.min);
              if (value.max) params.append(`filters[${key}][max]`, value.max);
            } else {
              params.append(`filters[${key}]`, value);
            }
          });
          
          return `${baseUrl}?${params.toString()}`;
        },
      
        async searchProperties() {
          const searchButton = document.querySelector(this.selectors.searchButton);
          if (!this.state.company?.id || !searchButton || searchButton.disabled) {
            return;
          }
          
          this._resetPaginationState();
          
          const filters = this._getFilterState();
          this.state.lastFilters = filters;
      
          this._updateOriginalPropertyDisplayWithFilters(filters);
          const originalButtonHTML = searchButton.innerHTML;
          const btnWidth = searchButton.offsetWidth;
          const btnHeight = searchButton.offsetHeight;
          const baseUrl = `https://imoveis.c2sapp.com/api/v1/company/${this.state.company.id}/properties`;
          const url = this._buildFilterUrl(baseUrl, filters, 1);
          
          $(searchButton).prop("disabled", true).css({ width: `${btnWidth}px`, height: `${btnHeight}px`, display: 'flex', alignItems: 'center', justifyContent: 'center', cursor: 'not-allowed' }).html('<div class="button-load"><div class="loader-indicator"></div></div>');
          
          try {
            const response = await fetch(url, { headers: { 'Authorization': this.constants.indicatorToken } });
            if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
            const data = await response.json();
      
            this.state.hasMorePages = !data.pagination.last_page;
            this._renderProperties(data.properties || []);
            this._toggleAdvancedSearch(false);
            this._switchResultsView('filtered');
      
            const iaFooter = document.querySelector('.ia-button-footer');
            const filterFooter = document.querySelector(this.selectors.buttonContainerFilter);
            const titleElement = document.querySelector('.modal-title-indicator h1');
      
            if (this.state.isAiFlow) {
              if (iaFooter) iaFooter.style.display = 'flex';
              if (filterFooter) filterFooter.style.display = 'none';
              if (titleElement) titleElement.textContent = `Envie sugestões de imóveis para sua mensagem de IA ser mais assertiva.`;
            } else {
              if (iaFooter) iaFooter.style.display = 'none';
              if (filterFooter) filterFooter.style.display = 'flex';
              if (titleElement) titleElement.textContent = `Imóveis semelhantes`;
            }
      
          } catch (error) {
            console.error(error);
            this._renderProperties([]);
            document.querySelector(this.selectors.similarText).innerHTML = `Erro ao buscar propriedades. Tente novamente.`;
            this._toggleAdvancedSearch(false);
            this._switchResultsView('filtered');
          } finally {
            $(searchButton).prop("disabled", false).html(originalButtonHTML).css({ width: '', height: '', display: '', alignItems: '', justifyContent: '', cursor: 'pointer' });
            this._validatePriceRange();
          }
        },
      
        _handleScroll(event) {
          const { scrollTop, scrollHeight, clientHeight } = event.target;
          const isNearBottom = scrollTop + clientHeight >= scrollHeight - 200;
      
          if (isNearBottom && !this.state.isLoading && this.state.hasMorePages) {
            this._loadMoreProperties();
          }
        },
      
        async _loadMoreProperties() {
          this.state.isLoading = true;
          const loader = document.querySelector(this.selectors.scrollLoader);
          if (loader) loader.style.display = 'flex';
          
          const nextPage = this.state.currentPage + 1;
          const baseUrl = `https://imoveis.c2sapp.com/api/v1/company/${this.state.company.id}/properties`;
          const url = this._buildFilterUrl(baseUrl, this.state.lastFilters, nextPage);
          
          try {
              const response = await fetch(url, { headers: { 'Authorization': this.constants.indicatorToken } });
              if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
              const data = await response.json();
              
              this.state.currentPage = data.pagination.current_page;
              this.state.hasMorePages = !data.pagination.last_page;
              
              this._renderProperties(data.properties || [], true); 
          } catch (error) {
              console.error(error);
              this.state.hasMorePages = false;
          } finally {
              this.state.isLoading = false;
              if (loader) loader.style.display = 'none';
          }
        },

        _formatNeighborhoodsText(neighborhoods) {
          if (!neighborhoods || neighborhoods.length === 0) {
            return '';
          }
          if (neighborhoods.length === 1) {
            return neighborhoods[0];
          }
          
          const firstNeighborhood = neighborhoods[0];
          const remainingCount = neighborhoods.length - 1;
          const badgeStyle = "display: inline-flex; align-items: center; justify-content: center; max-width: 40px; max-height: 40px; font-size: 14px; font-weight: 500; color: #333; background-color: #E0E0E0; border-radius: 50%; margin-left: 8px; vertical-align: middle; padding: 4px; cursor: pointer;";
          const badgeHtml = `<span class="neighborhood-badge" style="${badgeStyle}">+${remainingCount}</span>`;
      
          return `${firstNeighborhood}${badgeHtml}`;
        },
      
        _formatBedroomsText(bedroomsFilterValue) {
          if (!bedroomsFilterValue) return '-';
          const number = parseInt(bedroomsFilterValue, 10);
          if (isNaN(number)) return '-';
          return `${number} quarto(s)`;
        },
        
        _formatParkingSpotsText(parkingFilterValue) {
          if (!parkingFilterValue) return '-';
          if (parkingFilterValue.toLowerCase() === 'indiferente') {
            return `Indiferente`;
          }
          const number = parseInt(parkingFilterValue, 10);
          if (isNaN(number)) return '-';
          return `${number} vaga(s)`;
        },
      
        _formatAreaRangeText(min, max) {
          const format = (value) => Number(value).toLocaleString('pt-BR');
          if (min && max) return `${format(min)}m² - ${format(max)}m²`;
          if (min) return `${format(min)}m²+`;
          if (max) return `Até ${format(max)}m²`;
          return `-`;
        },
      
        _formatPriceRangeText(min, max) {
          const format = (value) => Number(value).toLocaleString('pt-BR');
          if (min && max) return `R$ ${format(min)} - R$ ${format(max)}`;
          if (min) return `R$ ${format(min)}+`;
          if (max) return `Até R$ ${format(max)}`;
          return `-`;
        },
      
        _formatTransactionTypesText(types) {
          if (!types || types.length === 0) {
            return 'Aluguel - Venda';
          }
          return types.map(type => type.charAt(0).toUpperCase() + type.slice(1)).join(' e ');
        }
      };

        function loadSimilarsData(productData, companyData, leadData) {
          const cacheKey = `${companyData.id}-${productData.prop_ref}`;
      
          // Se já tem dados em cache, usa eles
          if (cachedSimilarsData && cachedSimilarsData.key === cacheKey) {
            hideModalLoading();
            populateModalWithData(cachedSimilarsData.data, leadData);
            return;
          }
      
          const requestData = {
            lead_id: leadData.id,
            prop_ref: productData.prop_ref
          };
          const formattedInitials = formatNameToInitials(leadData?.customer?.name || customerName);
      
          $.ajax({
            type: "POST",
            url: `/integration/imovai/similars`,
            contentType: "application/json",
            data: JSON.stringify(requestData),
            success: function (data) {
              // Salva no cache
              cachedSimilarsData = { key: cacheKey, data: data };
      
              // Remove o loading e popula com os dados
              hideModalLoading();
              populateModalWithData(data, leadData);
            },
            error: function(xhr, status, error) {
              // Remove o loading e mostra busca avançada em caso de erro
              hideModalLoading();
              $(".container-indicator").hide();
              $(".advanced-search-content").show();

                $(".search-title").show();
                jumpButtonInFilter.show();

                jumpButtonInFilter.hide();
              }
            }
          });
        }

        function setupAdvancedSearchButton() {
          function extractNumericValue(priceText) {
            if (!priceText) return null;
      
            const numericValue = priceText
              .replace(/R\$\s*/g, '')
              .replace(/\./g, '')
              .replace(/,/g, '.')
              .trim();
      
            return parseFloat(numericValue) || null;
          }
      
          function fillPriceRangeInputs() {
            const priceInputs = document.querySelectorAll('.value-range input');
      
            if (priceInputs.length >= 2 && (priceInputs[0].value.trim() !== '' || priceInputs[1].value.trim() !== '')) {
              return;
            }
      
            const priceFilterElement = document.querySelector('.price-filter');
      
            if (!priceFilterElement) {
              return;
            }
      
            const priceText = priceFilterElement.textContent || priceFilterElement.innerText;
            const basePrice = extractNumericValue(priceText);
      
            if (!basePrice) {
              return;
            }
      
            const minPrice = Math.round(basePrice * 0.7);
            const maxPrice = Math.round(basePrice * 1.3);
      
            if (priceInputs.length >= 2) {
              if (window.FilterManager && window.FilterManager.formatMonetaryValue) {
                priceInputs[0].value = window.FilterManager.formatMonetaryValue(minPrice.toString());
                priceInputs[1].value = window.FilterManager.formatMonetaryValue(maxPrice.toString());
              } else {
                priceInputs[0].value = minPrice.toLocaleString('pt-BR');
                priceInputs[1].value = maxPrice.toLocaleString('pt-BR');
              }
            }
          }
      
          $(document).on('click', '.advanced-search', function() {
            setTimeout(() => {
              fillPriceRangeInputs();
            }, 150);
          });
        }

          function processContent(summary){
            if(!summary){ return defaultText; }
      
            var normalized = (typeof summary === 'string') ? summary : JSON.stringify(summary);
            var contentLines = normalized.split(/\r?\n/);
            var processedLines = contentLines.map(function(line){
              var trimmed = line.trim();
              if(trimmed.startsWith('- **')){
                var content = trimmed.substring(2);
                var boldContent = content.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
                return '<li>' + boldContent + '</li>';
              } else if(trimmed.startsWith('**')){
                var boldOnly = line.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
                return boldOnly;
              } else if(trimmed){
                return line;
              } else {
                return '';
              }
            });
      
            var joined = processedLines.join('\n');
            if(joined.indexOf('<li>') >= 0){
              joined = '<ul style="margin-left: 18px; padding-left: 18px;">' + joined + '</ul>';
            }
            return joined || defaultText;
          }
      
          function extractTitleAndBody(text){
            var title = 'Resumo Vetra';
            var body = text || '';
            if (body) {
              var match = String(body).trim().match(/^\*\*(.*?)\*\*/);
              if (match) {
                title = match[1];
                body = String(body).replace(/^\s*\*\*.*?\*\*\s*\r?\n?/, '');
              }
            }
            return { title: title, body: body };

      var makeLoginForm = function() {

        $(".title-modal").text('Recuperando senha');

        $(".back-to-login-button").show();
        $(".instructions").show();
      
        $(".login-form").hide();

      var setUsernameInDropdown = function() {
        let multiUserDropdown = $('#multi-user-dropdown')
      
        if ($('input[name="multi_user"]:checked').val()) {
          multiUserDropdown.text($('input[name="multi_user"]:checked').data('username'))
          multiUserDropdown.append('<span class="caret"></span>')
        }
      }
      
      var checkMultiLoginMessages = function() {
        if($('.success-message').data('message-success')) {
          toastr.success($('.success-message').data('message-success'));
        } else if ($('.error-message').data('message-error')) {
          toastr.error($('.error-message').data('message-error'));
        }
      }
      
      $(".back-to-login-button").click(function() {
        makeLoginForm();
      });
      
      $(".go-to-recovery-password-button").click(function() {
        makeRecoveryForm();
      });
      
      $(".add-account-session").click(function() {
        $('#multi-login-modal').modal('show');
      });
      
      $('input[name="multi_user"]').on('click', function () {
        let url = `/webapp/set_multi_user?id=${ $(this).val() }`
      
        $.ajax({
          url: url,
          type: 'GET',
          dataType: 'json',
          success: function(data) {
            if(data.success) {
              toastr.success('Usuário alterado com sucesso')
              window.location.reload();
            } else {
              toastr.error(data.message || "Erro ao mudar de usuário")
            }
          },
          error: function(data) {
            toastr.error(data.message || "Erro ao mudar de usuário")
          }
        });
      })

        makeLoginForm();
        setUsernameInDropdown();
        checkMultiLoginMessages();
      });

      $(document).ready(function(){
        $.get('/webapp/loader/cloud_humans', { cache: true }).done(function() {
          fetchMenu();
          onClickMenu();
          onClickNavbar();
        });

      function fetchMenu(){
        return $.get('/webapp/loader/menu', { controller_name: 'leads', action_name: 'show' }, function(data) {
          var script = document.createElement('script');
          script.type = 'text/javascript';
          script.text = data;
          document.body.appendChild(script);
      
          loadMenu();
          fetchBadges();
        });
      }
      
      function onClickMenu() {
        $('#side-menu').on('click', 'a', function(event) {
          try {      
            var linkTitle = $(this).attr('data-menu');
            var linkHref = $(this).attr('href');   // O valor do href do link
      
            if (!linkTitle) {
              linkTitle = $(this).find('span:first').text().trim() || $(this).text(); // O texto do link
            }
      
            // Chama a função para enviar os dados ao Mixpanel
            sendMixpanelData(linkTitle, linkHref, 'Menu Clicked');
            // Se quiser continuar com a navegação, não use preventDefault()
            // ou redirecione manualmente com window.location = linkHref;
          } catch (error) {
            // Captura e registra o erro no console
            console.error('Erro ao processar o clique no link:', error);
          }
        });
      }
      
      function onClickNavbar() {
        $('#multi-user-menu').on('click', 'a', function(event) {
          try {
            var linkTitle = $(this).data('menu') || $(this).text().trim();
            var linkHref = $(this).attr('href');
      
            sendMixpanelData(linkTitle, linkHref, 'Menu Navbar Clicked');
          } catch (error) {
            console.error('Erro ao processar o clique na navbar:', error);
          }
        });
      }
      
      function sendMixpanelData(title, linkTo, eventName) {
        var mixpanelData = {
          data: {
            title: title,
            link: linkTo,
          },
          event_name: eventName,
        };
      
        $.ajax({
          type: 'POST',
          url: '/mixpanel/create',
          data: JSON.stringify(mixpanelData),
          contentType: 'application/json',
          success: function(response) {
            // console.log('Mixpanel data sent successfully:', response);
          },
          error: function(error) {
            console.error('Error sending Mixpanel data:', error);
          }
        });
      }
