      function convertHMS(value) {
        const sec = parseInt(value, 10); // convert value to number if it's string
        let hours = Math.floor(sec / 3600); // get hours
        let minutes = Math.floor((sec - hours * 3600) / 60); // get minutes
        let seconds = sec - hours * 3600 - minutes * 60; //  get seconds
      
        // add 0 if value < 10; Example: 2 => 02
        if (hours < 10) {
          hours = "0" + hours;
        }
        if (minutes < 10) {
          minutes = "0" + minutes;
        }
        if (seconds < 10) {
          seconds = "0" + seconds;
        }
      
        return hours + ":" + minutes + ":" + seconds; // Return is HH : MM : SS
      }

      function updateBatchModalCounts() {
        const selectedCheckboxes = document.querySelectorAll('input[name="lead_ids[]"]:checked');
        const selectedCount = selectedCheckboxes.length;
        let sectionTotal = 0;
      
        // Detectar quantas seções distintas estão com leads marcados
        const sections = new Set();
        selectedCheckboxes.forEach(function(cb) {
          const leadGroup = cb.closest('.lead-group');
          if (leadGroup) sections.add(leadGroup.id);
        });
        const isMultiSection = sections.size > 1;
      
        // Ler o total da seção correspondente ao lead_list_type (a seção que será usada no scope "all")
        const leadListTypeEl = document.getElementById('lead-list-type');
        const leadListType = leadListTypeEl ? leadListTypeEl.value : null;
      
        if (leadListType) {
          const groupCheckbox = document.getElementById(leadListType);
          const sectionHeader = groupCheckbox ? groupCheckbox.closest('h2') : null;
          if (sectionHeader) {
            const match = sectionHeader.textContent.match(/\((\d+)\)/);
            if (match) sectionTotal = parseInt(match[1], 10);
          }
        }
      
        // Fallback: primeiro checkbox selecionado
        if (!sectionTotal && selectedCount > 0) {
          const leadGroup = selectedCheckboxes[0].closest('.lead-group');
          if (leadGroup) {
            const section = leadGroup.id.replace('lead-group-', '');
            const header = document.querySelector(`h2[data-section="${section}"]`);
            if (header) {
              const match = header.textContent.match(/\((\d+)\)/);
              if (match) sectionTotal = parseInt(match[1], 10);
            }
          }
        }
      
        // Fallback final: total da aba ativa
        if (!sectionTotal) {
          const activeTab = document.querySelector('.tab-pane.active');
          if (activeTab && activeTab.dataset.total) {
            sectionTotal = parseInt(activeTab.dataset.total, 10);
          }
        }
      
        // Atualiza no modal
        const selectedEl = document.getElementById('selected-count');
        const selectedCountEl = document.getElementById('scope_selected_count');
        const totalEl = document.getElementById('total-count');
        const totalCountEl = document.getElementById('scope_all_count');
      
        if (selectedEl) selectedEl.textContent = selectedCount;
        if (selectedCountEl) selectedCountEl.value = selectedCount;
        if (totalEl) totalEl.textContent = sectionTotal;
        if (totalCountEl) totalCountEl.value = sectionTotal;
      
        // Automação do radio e bloqueio de scope "all" em seleção multi-seção
        const radioSelected = document.querySelector('#scope_selected');
        const radioAll = document.querySelector('#scope_all');
      
        if (radioSelected && radioAll) {
          const pendingBulkUpdateDisabled = radioAll.dataset.pendingDisabled !== 'false';
          radioAll.disabled = isMultiSection || pendingBulkUpdateDisabled;
      
          if (isMultiSection) {
            radioSelected.checked = true;
          } else if (selectedCount > 0) {
            radioSelected.checked = true;
          } else {
            radioAll.checked = true;
          }
        }
      }

      var makeLoginForm = function() {
        $(".title-modal").text('Adicionando conta');
        $(".recovery-form").hide();
        $(".back-to-login-button").hide();
        $(".instructions").hide();
      
        $(".login-form").show();
        $(".go-to-recovery-password-button").show();
      }
      
      var makeRecoveryForm = function() {
        $(".title-modal").text('Recuperando senha');
        $(".recovery-form").show();
        $(".back-to-login-button").show();
        $(".instructions").show();
      
        $(".login-form").hide();
        $(".go-to-recovery-password-button").hide();
      }
      
      var setUsernameInDropdown = function() {
        let multiUserDropdown = $('#multi-user-dropdown')
      
        if ($('input[name="multi_user"]:checked').val()) {
          multiUserDropdown.text($('input[name="multi_user"]:checked').data('username'))
          multiUserDropdown.append('<span class="caret"></span>')
        }
      }
      
      var checkMultiLoginMessages = function() {
        if($('.success-message').data('message-success')) {
          toastr.success($('.success-message').data('message-success'));
        } else if ($('.error-message').data('message-error')) {
          toastr.error($('.error-message').data('message-error'));
        }
      }
      
      $(".back-to-login-button").click(function() {
        makeLoginForm();
      });
      
      $(".go-to-recovery-password-button").click(function() {
        makeRecoveryForm();
      });
      
      $(".add-account-session").click(function() {
        $('#multi-login-modal').modal('show');
      });
      
      $('input[name="multi_user"]').on('click', function () {
        let url = `/webapp/set_multi_user?id=${ $(this).val() }`
      
        $.ajax({
          url: url,
          type: 'GET',
          dataType: 'json',
          success: function(data) {
            if(data.success) {
              toastr.success('Usuário alterado com sucesso')
              window.location.reload();
            } else {
              toastr.error(data.message || "Erro ao mudar de usuário")
            }
          },
          error: function(data) {
            toastr.error(data.message || "Erro ao mudar de usuário")
          }
        });
      })
      
      $(document).ready(function() {
        makeLoginForm();
        setUsernameInDropdown();
        checkMultiLoginMessages();
      });
    </script>
    <script defer="defer">
      $(document).ready(function(){
        $.get('/webapp/loader/cloud_humans', { cache: true }).done(function() {
          fetchMenu();
          onClickMenu();
          onClickNavbar();
        });
      });
      
      function fetchMenu(){
        return $.get('/webapp/loader/menu', { controller_name: 'v2_leads', action_name: 'index' }, function(data) {
          var script = document.createElement('script');
          script.type = 'text/javascript';
          script.text = data;
          document.body.appendChild(script);
      
          loadMenu();
          fetchBadges();
        });
      }
      
      function onClickMenu() {
        $('#side-menu').on('click', 'a', function(event) {
          try {      
            var linkTitle = $(this).attr('data-menu');
            var linkHref = $(this).attr('href');   // O valor do href do link
      
            if (!linkTitle) {
              linkTitle = $(this).find('span:first').text().trim() || $(this).text(); // O texto do link
            }
      
            // Chama a função para enviar os dados ao Mixpanel
            sendMixpanelData(linkTitle, linkHref, 'Menu Clicked');
            // Se quiser continuar com a navegação, não use preventDefault()
            // ou redirecione manualmente com window.location = linkHref;
          } catch (error) {
            // Captura e registra o erro no console
            console.error('Erro ao processar o clique no link:', error);
          }
        });
      }
      
      function onClickNavbar() {
        $('#multi-user-menu').on('click', 'a', function(event) {
          try {
            var linkTitle = $(this).data('menu') || $(this).text().trim();
            var linkHref = $(this).attr('href');
      
            sendMixpanelData(linkTitle, linkHref, 'Menu Navbar Clicked');
          } catch (error) {
            console.error('Erro ao processar o clique na navbar:', error);
          }
        });
      }
      
      function sendMixpanelData(title, linkTo, eventName) {
        var mixpanelData = {
          data: {
            title: title,
            link: linkTo,
          },
          event_name: eventName,
        };
      
        $.ajax({
          type: 'POST',
          url: '/mixpanel/create',
          data: JSON.stringify(mixpanelData),
          contentType: 'application/json',
          success: function(response) {
            // console.log('Mixpanel data sent successfully:', response);
          },
          error: function(error) {
            console.error('Error sending Mixpanel data:', error);
          }
        });
      }
    </script>
